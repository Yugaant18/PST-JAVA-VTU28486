import java.util.*;
import java.util.stream.*;

public class Main {
    public static void main(String[] args) {
        List<Employee> employees = Employee.sampleData();

        List<Employee> sorted = employees.stream()
                .sorted(Comparator.comparing(e -> e.department))
                .collect(Collectors.toList());

        sorted.forEach(System.out::println);
    }
}

class Employee {
    String name;
    String department;

    Employee(String name, String department) {
        this.name = name;
        this.department = department;
    }

    static List<Employee> sampleData() {
        return Arrays.asList(
            new Employee("Yugaant", "AI"),
            new Employee("Rahul", "HR"),
            new Employee("Arun", "Development"),
            new Employee("Priya", "AI")
        );
    }

    @Override
    public String toString() {
        return name + " - " + department;
    }
}
